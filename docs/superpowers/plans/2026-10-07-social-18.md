**Goal:** A single-day golf gathering a host can announce, invite people to, collect accept/decline replies for (waiting room), then turn into a normal Game Day built from the accepted players, with optional money tracking and per-event chat.

**Architecture:** Two new tables (`social_events`, `social_event_players`) plus small additive columns on `direct_messages` and `messages`. A pure logic module (`src/lib/socialEvents.ts`, unit-tested) and a Supabase access module (`src/lib/socialEventsApi.ts`). New screens under `app/(app)/social18/`. The day itself reuses `games/new.tsx` (with two new route params) so scoring, Stableford side game, spectating and Season ingestion all work unchanged.

**Tech Stack:** Expo SDK 54 / expo-router, React Native, Supabase (Postgres + RLS), jest-expo (`npm test`), TypeScript. Docs: this repo pins SDK 54 (see memory `project_expo_version`), NOT the version AGENTS.md mentions — use v54 docs.

```ts
// Social 18 — pure logic (no Supabase, no React), so it's unit-testable.
export type SocialStatus = 'invited' | 'accepted' | 'declined';
export type MoneyMethod = 'collector' | 'winners' | 'none';

export interface SocialPlayerRow { player_id: string; status: SocialStatus; paid: boolean }
export interface Due { toPlayerId: string; amount: number }

export const PRIZE_SPLITS: { label: string; value: number[] }[] = [
  { label: 'Winner takes all', value: [100] },
  { label: '60 / 40', value: [60, 40] },
  { label: '50 / 30 / 20', value: [50, 30, 20] },
  { label: '40 / 30 / 20 / 10', value: [40, 30, 20, 10] },
];

export function countByStatus(rows: SocialPlayerRow[]) {
  const out = { accepted: 0, invited: 0, declined: 0 };
  for (const r of rows) out[r.status]++;
  return out;
}

// Declined players never appear in the waiting room.
export function waitingRoomRows<T extends SocialPlayerRow>(rows: T[]): T[] {
  return rows.filter(r => r.status !== 'declined');
}

export function acceptedIds(rows: SocialPlayerRow[]): string[] {
  return rows.filter(r => r.status === 'accepted').map(r => r.player_id);
}

export function canStartDay(acceptedCount: number): boolean {
  return acceptedCount >= 2;
}

export function potOf(fee: number, acceptedCount: number): number {
  return fee * acceptedCount;
}

export function placeAmounts(fee: number, acceptedCount: number, split: number[]): number[] {
  const pot = potOf(fee, acceptedCount);
  return split.slice(0, acceptedCount).map(pct => (pot * pct) / 100);
}

// Who this player currently owes, and how much. Same direct-settlement rule
// as Swindle: each non-winner pays fee x pct to the winner of that place; a
// winner never pays themselves. Nothing is owed in "winners" mode until the
// event is complete (before that, "winners" are meaningless).
export function paymentsDue(a: {
  method: MoneyMethod; fee: number; split: number[]; ranking: string[];
  playerId: string; collectorId: string | null; complete: boolean;
}): Due[] {
  if (a.method === 'none' || a.fee <= 0) return [];
  if (a.method === 'collector') {
    return !a.complete && a.collectorId && a.collectorId !== a.playerId
      ? [{ toPlayerId: a.collectorId, amount: a.fee }]
      : [];
  }
  if (!a.complete) return [];
  return a.split
    .map((pct, i) => ({ toPlayerId: a.ranking[i], amount: (a.fee * pct) / 100 }))
    .filter(d => d.toPlayerId && d.toPlayerId !== a.playerId);
}

export function inviteContent(a: { eventName: string; dateLabel: string; teeTime: string | null; course: string | null }): string {
  const when = `📅 ${a.dateLabel}${a.teeTime ? ` at ${a.teeTime}` : ''}`;
  const lines = [`You've been invited to ${a.eventName}.`, when];
  if (a.course) lines.push(`⛳ ${a.course}`);
  return lines.join('\n');
}

export function chatAnnouncement(a: { hostFirstName: string; eventName: string; dateLabel: string; course: string | null }): string {
  return `${a.hostFirstName} is playing ${a.eventName} on ${a.dateLabel}${a.course ? ` at ${a.course}` : ''}. Who wants in?`;
}
```

```ts
import { supabase } from './supabase';
import { sendPushNotification } from './notifications';
import { inviteContent, chatAnnouncement, MoneyMethod, SocialPlayerRow } from './socialEvents';

export interface SocialEvent {
  id: string; society_id: string; host_player_id: string; name: string;
  event_date: string; tee_time: string | null; course_name: string | null;
  entry_fee: number; currency: string; money_method: MoneyMethod;
  collector_player_id: string | null; prize_split: number[];
  status: 'open' | 'started' | 'complete' | 'cancelled'; day_id: string | null;
}
export interface SocialPlayer extends SocialPlayerRow {
  id: string; source: 'invite' | 'chat'; display_name: string; avatar_url: string | null;
}
export interface NewSocialFields {
  name: string; eventDate: string; teeTime: string | null; courseName: string | null;
  entryFee: number; currency: string; moneyMethod: MoneyMethod;
  collectorId: string | null; prizeSplit: number[];
}
export interface NewSocialEvent extends NewSocialFields {
  societyId: string; hostId: string; hostName: string;
  inviteeIds: string[]; postToChat: boolean;
}

const EVENT_COLS = 'id,society_id,host_player_id,name,event_date,tee_time,course_name,entry_fee,currency,money_method,collector_player_id,prize_split,status,day_id';

export function formatEventDate(iso: string): string {
  const d = new Date(`${iso}T12:00:00`);
  return d.toLocaleDateString('en-GB', { weekday: 'short', day: 'numeric', month: 'short' });
}

export async function getMyPlayer(): Promise<{ id: string; display_name: string } | null> {
  const { data: { user } } = await supabase.auth.getUser();
  if (!user) return null;
  const { data } = await supabase.from('players').select('id, display_name').eq('auth_uid', user.id).maybeSingle();
  return (data as any) ?? null;
}

export async function createSocialEvent(n: NewSocialEvent): Promise<string> {
  const { data: ev, error } = await supabase.from('social_events').insert({
    society_id: n.societyId, host_player_id: n.hostId, name: n.name.trim(),
    event_date: n.eventDate, tee_time: n.teeTime, course_name: n.courseName,
    entry_fee: n.entryFee, currency: n.currency, money_method: n.moneyMethod,
    collector_player_id: n.moneyMethod === 'collector' ? (n.collectorId ?? n.hostId) : null,
    prize_split: n.prizeSplit,
  }).select(EVENT_COLS).single();
  if (error || !ev) throw error ?? new Error('Could not create event');
  const event = ev as unknown as SocialEvent;

  const rows = [
    { event_id: event.id, player_id: n.hostId, status: 'accepted', source: 'invite' },
    ...n.inviteeIds.filter(id => id !== n.hostId).map(id => ({ event_id: event.id, player_id: id, status: 'invited', source: 'invite' })),
  ];
  const { error: pErr } = await supabase.from('social_event_players').insert(rows);
  if (pErr) console.error('[social18] adding players failed', pErr);

  await sendInvites(event, n.hostId, n.inviteeIds);
  if (n.postToChat) await postAnnouncement(event, n.hostId, n.hostName);
  return event.id;
}

export async function updateSocialEvent(id: string, p: Partial<NewSocialFields>): Promise<void> {
  const patch: Record<string, unknown> = {};
  if (p.name !== undefined) patch.name = p.name.trim();
  if (p.eventDate !== undefined) patch.event_date = p.eventDate;
  if (p.teeTime !== undefined) patch.tee_time = p.teeTime;
  if (p.courseName !== undefined) patch.course_name = p.courseName;
  if (p.entryFee !== undefined) patch.entry_fee = p.entryFee;
  if (p.currency !== undefined) patch.currency = p.currency;
  if (p.moneyMethod !== undefined) patch.money_method = p.moneyMethod;
  if (p.collectorId !== undefined) patch.collector_player_id = p.collectorId;
  if (p.prizeSplit !== undefined) patch.prize_split = p.prizeSplit;
  const { error } = await supabase.from('social_events').update(patch).eq('id', id);
  if (error) throw error;
}

// Accept / decline (inbox card, chat "I'm in", event page "Can't make it").
// Inserts the row if the player wasn't invited (chat join), else updates it.
export async function respondToSocialEvent(
  event: Pick<SocialEvent, 'id' | 'name' | 'host_player_id'>,
  playerId: string, playerName: string, response: 'accepted' | 'declined',
): Promise<void> {
  const { data: existing } = await supabase.from('social_event_players')
    .select('id').eq('event_id', event.id).eq('player_id', playerId).maybeSingle();
  const { error } = existing
    ? await supabase.from('social_event_players').update({ status: response }).eq('id', (existing as any).id)
    : await supabase.from('social_event_players').insert({ event_id: event.id, player_id: playerId, status: response, source: 'chat' });
  if (error) throw error;

  if (playerId === event.host_player_id) return;
  const who = playerName.split(' ')[0];
  const content = response === 'accepted'
    ? `✅ ${who} is in for ${event.name}.`
    : `❌ ${who} can't make ${event.name}.`;
  // Best effort — must never undo the player's own reply.
  try {
    await supabase.from('direct_messages').insert({ sender_id: playerId, recipient_id: event.host_player_id, content, message_type: 'text' });
    sendPushNotification(response === 'accepted' ? `${who} is in!` : `${who} can't make it`, content, [event.host_player_id], { type: 'message', senderId: playerId });
  } catch (e) { console.error('[social18] host notify failed', e); }
}

// Stableford points per player across the day's matches, best first.
export async function fetchDayRanking(dayId: string): Promise<string[]> {
  const { data: matches } = await supabase.from('matches').select('id').eq('day_id', dayId).neq('status', 'cancelled');
  const ids = (matches ?? []).map((m: any) => m.id as string);
  if (!ids.length) return [];
  const totals: Record<string, number> = {};
  let from = 0;
  for (;;) {
    const { data } = await supabase.from('match_holes').select('player_id, stableford_pts')
      .in('match_id', ids).order('id').range(from, from + 999);
    const rows = (data ?? []) as any[];
    for (const r of rows) totals[r.player_id] = (totals[r.player_id] ?? 0) + (r.stableford_pts ?? 0);
    if (rows.length < 1000) break;
    from += 1000;
  }
  return Object.entries(totals).sort((a, b) => b[1] - a[1]).map(([id]) => id);
}
```

```tsx
                <Image source={resolveAvatar(p.player_id, p.avatar_url)} style={{ width: 32, height: 32, borderRadius: 16 }} />

                )}
              </View>
            );
          })}
        </View>

        {/* Invitee / joiner controls */}
        {!isHost && event.status === 'open' && (!mine || mine.status !== 'accepted') && me && (
          <TouchableOpacity style={[st.redBtn, busy && st.disabled]} disabled={busy} activeOpacity={0.85}
            onPress={() => run(() => respondToSocialEvent(event, me.id, me.display_name, 'accepted'))}>
            <Text style={st.redText}>I'm in</Text>
          </TouchableOpacity>
        )}
        {!isHost && event.status === 'open' && mine?.status === 'accepted' && me && (
          <TouchableOpacity style={[st.ghostBtn, busy && st.disabled]} disabled={busy} activeOpacity={0.85}
            onPress={() => run(() => respondToSocialEvent(event, me.id, me.display_name, 'declined'))}>
            <Text style={st.ghostText}>Can't make it</Text>
          </TouchableOpacity>
        )}

        {(mine?.status === 'accepted' || isHost) && event.status !== 'cancelled' && (
          <TouchableOpacity style={st.ghostBtn} onPress={() => router.push(`/(app)/social18/chat/${event.id}` as any)} activeOpacity={0.85}>
            <Text style={st.ghostText}>Event chat</Text>
          </TouchableOpacity>
        )}
        {event.day_id && (
          <TouchableOpacity style={st.primaryBtn} onPress={() => router.push(`/(app)/score/day/${event.day_id}` as any)} activeOpacity={0.85}>
            <Text style={st.primaryText}>Open the Game Day</Text>
          </TouchableOpacity>
        )}

        {/* Host-only controls — invitees never see these */}
        {isHost && event.status !== 'cancelled' && (
          <>
            {event.status === 'open' && (
              <TouchableOpacity
                style={[st.redBtn, !canStartDay(accepted.length) && st.disabled]}
                disabled={!canStartDay(accepted.length)} activeOpacity={0.85}
                onPress={() => router.push(
                  `/(app)/games/new?socialEventId=${event.id}&onlyPlayerIds=${accepted.join(',')}${event.course_name ? `&course=${encodeURIComponent(event.course_name)}` : ''}` as any,
                )}>
                <Text style={st.redText}>{canStartDay(accepted.length) ? `Start the day (${accepted.length} in)` : 'Need at least 2 players in'}</Text>
              </TouchableOpacity>
            )}
            {event.status === 'open' && (
              <>
                <TouchableOpacity style={st.ghostBtn} onPress={openInvite} activeOpacity={0.85}><Text style={st.ghostText}>Invite more</Text></TouchableOpacity>
                <TouchableOpacity style={st.ghostBtn} onPress={() => router.push(`/(app)/social18/create?id=${event.id}` as any)} activeOpacity={0.85}><Text style={st.ghostText}>Edit event</Text></TouchableOpacity>
              </>
            )}
            {event.status === 'started' && (
              <TouchableOpacity style={st.primaryBtn} onPress={() => setConfirm('complete')} activeOpacity={0.85}>
                <Text style={st.primaryText}>Mark complete</Text>
              </TouchableOpacity>
            )}
            {!complete && (
              <TouchableOpacity style={st.ghostBtn} onPress={() => setConfirm('cancel')} activeOpacity={0.85}>
                <Text style={[st.ghostText, { color: RED }]}>Cancel event</Text>
              </TouchableOpacity>
            )}
          </>
        )}
      </ScrollView>

      <PlayerSelectorSheet visible={showInvite} title="Tap to invite" players={roster}
        onSelect={p => { setRoster(prev => prev.filter(x => x.id !== p.id)); run(() => inviteMore(event, me?.display_name ?? '', [p.id])); }}
        onClose={() => setShowInvite(false)} />

      <ConfirmDialog visible={!!confirmCopy} title={confirmCopy?.title ?? ''} message={confirmCopy?.message ?? ''}
        confirmLabel={confirmCopy?.label ?? ''} destructive={confirmCopy?.destructive}
        onConfirm={onConfirm} onCancel={() => setConfirm(null)} />
    </View>
  );
}
```
